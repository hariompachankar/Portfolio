function validateInput(input) {
  const patterns = {
    email: /^[^\s@]+@[^\s@]+\.[^\s@]+$/,
    url: /^https?:\/\/.+/
  };
  return Object.entries(patterns)
    .every(([_, regex]) => regex.test(input));
}
